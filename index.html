/* ========================================================================== Backend API Integration (Real M2M Gateway Connection) ========================================================================== */ // ★ここをご自身のバックエンドAPIのRender URLに書き換えてください const BACKEND_API_URL = 'https://YOUR-BACKEND-SERVICE.onrender.com'; async function fetchBackendStatus() { try { const response = await fetch(`${BACKEND_API_URL}/`); if (response.ok) { const data = await response.json(); // ステータスバッジを「ONLINE」に更新 const badge = document.querySelector('.status-badge'); if (badge) { badge.innerText = `PROTOCOL: ONLINE [${data.protocol}]`; badge.style.color = '#00ff88'; badge.style.borderColor = 'rgba(0, 255, 136, 0.4)'; } // ログストリームにAPI接続成功ログを差し込む addLogEntryFromAPI('System', `Connected to Gateway: ${data.status}`, 'SUCCESS'); } } catch (error) { console.error('Backend connection error:', error); const badge = document.querySelector('.status-badge'); if (badge) { badge.innerText = `PROTOCOL: OFFLINE (RETRYING...)`; badge.style.color = '#ff4444'; } } } function addLogEntryFromAPI(agent, action, statusText) { const container = document.getElementById('log-stream'); if (!container) return; const entry = document.createElement('div'); entry.className = 'log-entry'; const time = new Date().toISOString().substr(11, 8); entry.innerHTML = ` ${time} [${agent}] ${action} ${statusText} `; container.insertBefore(entry, container.firstChild); if (container.children.length > 25) { container.removeChild(container.lastChild); } } // ページ読み込み時および10秒ごとにバックエンドの状態を取得 fetchBackendStatus(); setInterval(fetchBackendStatus, 10000);
